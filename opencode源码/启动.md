```ts

  //type TerminatedPayload = { code: number | null; signal: number | null }
  const exit = new Promise<TerminatedPayload>((resolve) => {
    //如果子进程提前退出了，就调用resolve回调函数，这样exit就是一个resolved状态的promis
    child.on("exit", (code: number | null, signal: NodeJS.Signals | null) => {
      console.log(`[cli] Process exited with code: ${code}, signal: ${signal}`)
      //通过await或者then回调函数可以获取到resolve传的值
      resolve({ code: code ?? null, signal: null })
    })
    //如果出错
    //then回调函数可以接收两个参数，这两个参数分别对应resolved状态的回调和rejected状态的回调，resolved状态就会执行第一个参数，rejected状态执行第二个参数
    //如果只传一个参数，默认是resolved状态的回调
    //这里没有通过resolve或者reject回调改变Promise状态，所以如果子进程有错误触发了这个事件的话，那么只是会通过EventEmitter触发error事件，而不会改变这个Promise的状态，所以进程如果只触发error事件，那么这个Promise永远是pending状态

      //在这里抛出错误不会改变exit这个Promise的状态
      console.error(`[cli] Process error: ${error.message}`)
      //这里通过EventEmitter触发error事件
      events.emit("error", error.message)
    })
    //在这个函数体中抛出错误会改变exit这个Promise的状态为rejected
  })

  const stdout = child.stdout
  const stderr = child.stderr

  if (stdout) {
    readline.createInterface({ input: stdout }).on("line", (line: string) => {
      //handleSqliteProgress中会触发sqlite事件
      if (handleSqliteProgress(events, line)) return
      events.emit("stdout", `${line}\n`)
    })
  }

  if (stderr) {
    readline.createInterface({ input: stderr }).on("line", (line: string) => {
      //handleSqliteProgress中会触发sqlite事件
      if (handleSqliteProgress(events, line)) return
      events.emit("stderr", `${line}\n`)
    })
  }
  //exit如果是resolved状态会触发这个then回调函数
  exit.then((payload) => {
    //触发terminated事件
    events.emit("terminated", payload)
  })
 
  const kill = () => {
    if (!child.pid) return
    treeKill(child.pid)
  }
  //返回创建的EventEmiter和杀死子进程的方法和子进程退出时的方法
   //这里利用了Node.js的闭包使得当前进程的pid可以保存于内存中,所以kill可以用
   //同时返回了events这个EventEmitter，所以外部在创建进程的时候可以调用on绑定事件及处理函数，这样每一个通过这个spawnCommand创建的函数都可以监听它需要的事件
```

```ts
//延迟Promise模式，提前创建Promise，把resolve/reject函数暴露到外面，让你能在任意时机、任意位置手动决定这个Promise何时兑现，这里通过闭包把它存到变量里返回出去
//普通的Promise只能executor中访问到resolve或者reject
// 接收一个泛型
function defer<T>() {
  /**
   * 要特别区分的是，栈上消亡的只是这个函数的执行上下文，也就是栈帧，它只是一个临时的执行现场；而真正被闭包捕获的变量并不住在栈上，它们被存放在堆上的词法环
   * 境里，也就是 V8 的 ContextObject。栈帧内部确实有一个 context 指针指向这个 ContextObject，但那个指针的职责只是让函数在执行期间能解析和访问这些闭包变
   * 量，它随栈帧一起销毁，并不负责让对象活下去。真正让 ContextObject 活下来的靠山，是返回对象 {promise, resolve, reject} 被外部变量（比如 
   * serverReady）持有时形成的堆上引用链。所以这里存在两个不同角色的引用：栈内的引用负责"用"，在函数执行时去读写闭包变量；堆上的引用负责"活"，让词法环境在
   * 函数返回之后依然存活。而垃圾回收机制判断一个对象要不要被回收，看的既不是栈帧还在不在，也不是闭包还在不在，而是从 GC 的根对象出发沿着引用链去标记可达性
   * ——serverReady 是模块级变量，本身就是根，又被渲染进程的 await 从另一头牵住，所以这条引用链从根一路可达，被标记为存活，GC 也就不会回收它；只有当所有引
   * 用都被释放、链条彻底断开之后，这簇对象才会在下一轮垃圾回收里被整体清除。把这些串起来就是一句话：栈帧负责执行，堆上的词法环境负责存活，栈内的引用管"用"，堆上的引用管"活"，而垃圾回收只认"从根是否可达"。
   */
  //构造resolve和reject这两个改变状态的函数
  let resolve!: (value: T) => void
  let reject!: (error: Error) => void
  //创建一个新Promise
  const promise = new Promise<T>((res, rej) => {
    resolve = res
    reject = rej
  })
  //通过变量返回创建的promise和它的改变状态的函数
  return { promise, resolve, reject }
}

export type ServerReadyData = {
  url: string
  username: string | null
  password: string | null
}
```

接下来的代码中调用了setInitStep({ phase: "sqlite_waiting" })，我们从这个函数开始捋一捋。
首先在初始化之前会先执行这个函数registerIpcHandlers。IPC顾名思义是用于进程间通信的，如果有了解安卓的可以和安卓类比，安卓的主线程不能用来做耗时任务，耗时任务需要再开一个线程来做，做完之后通过Handler来通知主线程更新渲染。
对于electron也是一样的，主进程和渲染进程都不做耗时任务，通过子进程和worker来做耗时任务，等任务结束了通过进程间通信来更新渲染或者是其他通信。
registerIpcHandlers就是用来注册不同的进程消息的处理函数，它接收一个对象参数，这个参数需要包含所有opencode定义的消息的处理函数，然后在registerIpcHandlers中使用ipcMain.handle统一注册给主进程，其他进程通过invoke或者是send和主进程之间通信。
```ts
//InitStep只有以下几种取值
export type InitStep = { phase: "server_waiting" } | { phase: "sqlite_waiting" } | { phase: "done" }
//刚打开桌面端opencode时初始化
let initStep: InitStep = { phase: "server_waiting" }
//EventEmitter我们前面已经学习过了
const initEmitter = new EventEmitter()
//改变当前阶段并保存于内存中，并且触发step事件，通知所有绑定这个事件的执行处理函数，这里需要注意如果没有绑定step事件，那么step事件就会被忽略
function setInitStep(step: InitStep) {
  initStep = step
  logger.log("init step", { step })
  initEmitter.emit("step", step)
}
registerIpcHandlers({
  ...
  //走到这里或许会疑惑只有invoke或者send了await-initialization进程级别事件，这个处理函数awaitInitialization才会被触发，这个时候才会注册step事件。那如果在触发await-initialization之前就已经发送了step事件怎么办
  //我们可以看到awaitInitialization接收一个参数sendStep函数，通过这个函数将当前的阶段发送给渲染进程。
  /**
     * sendstep在registerIpcHandlers中被传入，下面是registerIpcHandlers注册事件时的调用，这里的deps是这里传入的对象
     * ipcMain.handle("await-initialization", 
     * (event: IpcMainInvokeEvent) => {
     * 主进程向发起这次IPC请求的渲染进程发送消息，
     * IpcMainInvokeEvent是事件对象，这里面会包含哪一个进程触发的invoke，
     * 然后这里获取到之后再调用send事件将现阶段信息发送回去
     * const send = (step: InitStep) => event.sender.send("init-step", step)
     * 然后调用我们注册时的处理函数，并把send函数作为传入
     * return deps.awaitInitialization(send)
     * })
     */
  //然后在渲染进程中会监听init-step事件，然后在该进程内保存阶段
  /**
   * awaitInitialization: (onStep) => {
   * const handler = (_: unknown, step: InitStep) => onStep(step)
   * ipcRenderer.on("init-step", handler)
   * 然后向主线程发送await-initialization事件获取到服务器信息
   * return ipcRenderer.invoke("await-initialization").finally(() => {
   * 最后移除监听
   * ipcRenderer.removeListener("init-step", handler)
   * })
   * }
   */
  //渲染进程调用的时候，会将init-step传入的阶段信息，保存起来
  //window.api.awaitInitialization((next) => setStep(next as InitStep)).catch(() => undefined)
  //回到我们的问题，在触发await-initialization之前就已经发送了step事件，那么这个时候step事件肯定会被忽略。但是会在主进程内存当中通过initStep变量保存当前阶段。然后后续渲染进程想要触发awaitInitialization事件，会调用window.api.awaitInitialization函数，将它的保存阶段的函数传入，同时注册init-step监听。我们的主进程收到这个事件时，会通过IpcMainInvokeEvent事件对象获取到这个事件的发送方，然后触发init-step事件，将当前的状态发送给渲染进程，这样就做到了双边阶段的同步。如果渲染进程在阶段改变的间隙触发了await-initialization事件，那么当前阶段信息会通过init-step事件被发送给渲染进程，但是在阶段改变之后触发的，因为阶段被保存于一个变量中，所以上一个阶段被覆盖，那么渲染进程就不会拿到上一个阶段的信息。

  awaitInitialization: async (sendStep) => {
    sendStep(initStep)
    //这个回调函数会在initEmitter通过emit发送step事件的时候触发，再次调用sendStep将当前阶段通知给渲染进程，然后渲染进程更新渲染
    const listener = (step: InitStep) => sendStep(step)
    initEmitter.on("step", listener)
    try {
      logger.log("awaiting server ready")
      /** 前面有提到过这个延迟promise，通过await得到的是resolve的参数，也就是之前传入的服务器信息
       * serverReady.resolve({ url,username: "opencode",password,})
       */
      const res = await serverReady.promise
      logger.log("server ready", { url: res.url })
      return res
    } finally {
      //最后移除step事件的监听
      initEmitter.off("step", listener)
    }
  },
  ...
})
```

```ts
      //这里的作用是等待数据库初始化完成，如果没有完成await会使得代码一直停留在这里，知道promise的状态改变
      await sqliteDone?.promise
    }
    //health是spawnLocalServer返回的，wait中保存了服务器进程是否出错还是健康的结果
    //这里会等待3秒服务器健康检查的结果，如果3秒内wait的状态还没有改变就抛出错误，服务器启动失败了
    await Promise.race([
      health.wait,
      //这里的 _ 是 数字分隔符，只用于让数字更容易读
      delay(30_000).then(() => {
        throw new Error("Sidecar health check timed out")
      }),
    ]).catch((error) => {
      logger.error("sidecar health check failed", error)
    })

    logger.log("loading task finished")
  })()

  const globals = {
    updaterEnabled: UPDATER_ENABLED,
    deepLinks: pendingDeepLinks,
  }
  //如果需要迁移数据库，等待1s看loadingTask的状态有没有发生改变，发生改变了就说明数据库肯定初始化成功了，因为初始化/迁移不成功loadingTask会一直是pending状态不会向下执行代码
  if (needsMigration) {
    //如果1s内没有成功就显示一个加载中窗口
    const show = await Promise.race([loadingTask.then(() => false), delay(1_000).then(() => true)])
    if (show) {
      overlay = createLoadingWindow(globals)
      await delay(1_000)
    }
  }
  //等待数据库初始化成功和服务器健康检查结果
  await loadingTask
  //更新阶段状态
  setInitStep({ phase: "done" })
  //如果加载中不为空，就要等待加载窗口渲染完成，这里也是使用了一个演示promise，用来阻塞当前代码
  //不完成不能向下走
  if (overlay) {
    await loadingComplete.promise
  }
  //开启主窗口和渲染菜单
  mainWindow = createMainWindow(globals)
  wireMenu()
  //最后关闭加载窗口
  overlay?.close()
```
