2、在打开客户端的时候，packages\desktop-electron会调用initialize方法，在这里面首先会查找是否设置了固定的端口环境变量process.env.OPENCODE_PORT。
如果没有设置会动态获取一个可用端口
```ts
//这里await了就会返回port，而不是一个resolved状态的promise
return await new Promise<number>((resolve, reject) => {
    //创建一个服务器
    const server = createServer()
    //出错直接转为rejected状态
    server.on("error", reject)
    //这里不是监听0号端口，而是让操作系统在本机自动挑选一个可用的端口
    server.listen(0, "127.0.0.1", () => {
        //获取到这个可用的端口信息
        //一般是这样的
        /**
         * {
         * address: "127.0.0.1",
         * family: "IPv4",
         * port: 54321}
         */
      const address = server.address()
      if (typeof address !== "object" || !address) {
        //如果没有，就报错
        server.close()
        reject(new Error("Failed to get port"))
        return
      }
      //获取到端口号
      const port = address.port
      //获取成功，关闭当前这个获取动态端口的服务，这里resolve之后表示当前这个promise转为resolved状态
      server.close(() => resolve(port))
    })
  })
```

之后将合成的envs传给buildCommand创建开启服务器的bash指令
```ts
function buildCommand(args: string, env: Record<string, string>) {
  //在桌面端设置wsl可用是在调用ElectronAPI的setWslConfig时设置的，但是触发这个的函数被注释掉了，所以其实是不会设置的，所以这个分支不会进入。
  if (process.platform === "win32" && isWslEnabled()) {
    console.log(`[cli] Using WSL mode`)
    const version = app.getVersion()
    const script = [
      //命令失败时让脚本退出
      "set -e",
      //将 WSL 内的 CLI 路径设为 $HOME/.opencode/bin/opencode
      'BIN="$HOME/.opencode/bin/opencode"',
      //如果该路径下没有可执行文件，就从 https://opencode.ai/install 下载并运行安装脚本，
      // 安装与桌面应用相同版本的 CLI；
      // --no-modify-path 表示不修改 PATH。
      'if [ ! -x "$BIN" ]; then',
      `  curl -fsSL https://opencode.ai/install | bash -s -- --version ${shellEscape(version)} --no-modify-path`,
      //if的结束符
      "fi",
      //${envPrefix(env)}是获取到当前环境变量并转为字符串，${}用来获取变量值
      `${envPrefix(env)} exec "$BIN" ${args}`,
    ].join("\n")

    return { cmd: "wsl", cmdArgs: ["-e", "bash", "-lc", script] }
  }
  //像我的电脑是windows，所以会走这条路径
  if (process.platform === "win32") {
    //获取确定随桌面应用打包的 CLI 程序的路径，得到的路径类似D:\code\opencode\packages\desktop-electron\resources\opencode-cli.exe
    const sidecar = getSidecarPath()
    console.log(`[cli] Windows direct mode, sidecar: ${sidecar}`)
    return { cmd: sidecar, cmdArgs: args.split(" ") }
  }
  //非windows系统用bash命令启动
  const sidecar = getSidecarPath()
  const shell = process.env.SHELL || "/bin/sh"
  const line = shell.endsWith("/nu") ? `^\"${sidecar}\" ${args}` : `\"${sidecar}\" ${args}`
  console.log(`[cli] Unix mode, shell: ${shell}, command: ${line}`)
  return { cmd: shell, cmdArgs: ["-l", "-c", line] }
}

```

然后对这个子进程进行监听，这里通过Node.js的EventEmitter来管理事件的触发和处理，可以把EventEmitter看成一个事件总线，emit函数用来触发事件总线上绑定的事件，on则用来将事件绑定到事件总线上，如果想要某一个事件被触发，那么需要绑定这个事件，所以说下面这些代码中的事件不一定在当前这个子进程中都会触发
```ts
const events = new EventEmitter()
  const exit = new Promise<TerminatedPayload>((resolve) => {
    child.on("exit", (code: number | null, signal: NodeJS.Signals | null) => {
      console.log(`[cli] Process exited with code: ${code}, signal: ${signal}`)
      resolve({ code: code ?? null, signal: null })
    })
    child.on("error", (error: Error) => {
      console.error(`[cli] Process error: ${error.message}`)
      events.emit("error", error.message)
    })
  })

  const stdout = child.stdout
  const stderr = child.stderr

  if (stdout) {
    readline.createInterface({ input: stdout }).on("line", (line: string) => {
      //handleSqliteProgress中会触发sqlite事件
      if (handleSqliteProgress(events, line)) return
      events.emit("stdout", `${line}\n`)
    })
  }

  if (stderr) {
    readline.createInterface({ input: stderr }).on("line", (line: string) => {
      //handleSqliteProgress中会触发sqlite事件
      if (handleSqliteProgress(events, line)) return
      events.emit("stderr", `${line}\n`)
    })
  }

  exit.then((payload) => {
    events.emit("terminated", payload)
  })
  
  const kill = () => {
    if (!child.pid) return
    treeKill(child.pid)
  }
  //返回创建的EventEmiter和杀死子进程的方法和子进程退出时的方法
  //到这里开启子进程的函数spawnCommand就结束了
  return { events, child: { kill }, exit }
```

```ts
const wait = (async () => {
    const url = `http://${hostname}:${port}`

    const ready = async () => {
      while (true) {
        await new Promise((resolve) => setTimeout(resolve, 100))
        if (await checkHealth(url, password)) return
      }
    }

    const terminated = async () => {
      const payload = await exit
      throw new Error(
        `Sidecar terminated before becoming healthy (code=${payload.code ?? "unknown"} signal=${
          payload.signal ?? "unknown"
        })`,
      )
    }

    await Promise.race([ready(), terminated()])
  })()

  return { child, health: { wait }, events }
```
