<script lang="ts" setup>

import { ref } from 'vue'

const before = ref(true)

const activeNames = ref(['1'])

const loading = ref(false)

const beforeCollapse = (): Promise<boolean> => {

  loading.value = true

  return new Promise((resolve) => {

    setTimeout(() => {

      loading.value = false

      return resolve(before.value)

    }, 1000)

  })

}

</script>
